cmake_minimum_required(VERSION 3.16)
project(SpectralAudio VERSION 1.0.0 LANGUAGES CXX)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(spectral_core STATIC src/predictor.cpp src/model_registry.cpp src/stream.cpp generated/model_data.cpp)
set_target_properties(spectral_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(spectral_core PUBLIC include src generated)
if(MSVC)
  target_compile_options(spectral_core PRIVATE /O2 /W4 /fp:strict /utf-8)
else()
  target_compile_options(spectral_core PRIVATE -O3 -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(spectral_apo_processor_unit apo/processor_test.cpp)
  target_link_libraries(spectral_apo_processor_unit PRIVATE spectral_core)
  add_test(NAME processor_realtime_contract COMMAND spectral_apo_processor_unit)
  add_executable(spectral_model_probe tests/model_probe.cpp)
  target_link_libraries(spectral_model_probe PRIVATE spectral_core)
  add_library(spectral_verify SHARED tests/bridge.cpp)
  target_link_libraries(spectral_verify PRIVATE spectral_core)
endif()
if(WIN32)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Spectral Audio requires Windows x64")
  endif()
  add_library(SpectralAudio SHARED apo/apo.cpp apo/registration.cpp apo/SpectralAudio.def)
  set_target_properties(SpectralAudio PROPERTIES PREFIX "")
  add_executable(spectral_apo_runtime apo/runtime_cli.cpp)
  add_executable(spectral_apo_measure apo/measure.cpp)
  add_executable(spectral_apo_smoke apo/smoke.cpp)
  foreach(target SpectralAudio spectral_apo_runtime spectral_apo_measure spectral_apo_smoke)
    target_compile_definitions(${target} PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
    target_link_libraries(${target} PRIVATE spectral_core ole32 uuid advapi32 ksuser)
    if(MSVC)
      target_compile_options(${target} PRIVATE /O2 /W4 /utf-8 /fp:strict)
    else()
      target_compile_options(${target} PRIVATE -O3 -fno-fast-math -ffp-contract=off)
      target_link_options(${target} PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
  endforeach()
  if(MINGW)
    target_link_options(spectral_apo_runtime PRIVATE -municode)
    if(TARGET spectral_apo_processor_unit)
      target_link_options(spectral_apo_processor_unit PRIVATE -static -static-libgcc -static-libstdc++)
      target_link_options(spectral_model_probe PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
  endif()
  if(BUILD_TESTING)
    add_test(NAME windows_apo_abi COMMAND spectral_apo_smoke $<TARGET_FILE:SpectralAudio>)
  endif()
endif()
